#include <WiFi.h>
#include <PubSubClient.h>
#include <DHT.h>
#include <ArduinoJson.h>

// ── Configuration ──────────────────────────────────────────────
const char* WIFI_SSID     = "Wokwi-GUEST";
const char* WIFI_PASSWORD = "";
const char* DEVICE_TOKEN  = "YOUR_DEVICE_TOKEN_HERE";
const char* MQTT_BROKER   = "test.mosquitto.org";
const int   MQTT_PORT     = 1883;
const char* MQTT_TOPIC    = "iot/gdpr/telemetry";
const char* DEVICE_ID     = "esp32-wokwi-1";

// DHT22 sensor on GPIO 4
#define DHTPIN  4
#define DHTTYPE DHT22

// Simulated sensors (ADC pins)
#define HR_PIN     34
#define SPO2_PIN   35
#define RESP_PIN   32   // respiratory rate potentiometer (simulate)
#define BPSYS_PIN  33   // blood pressure systolic (simulate)
#define BPDIA_PIN  25   // blood pressure diastolic (simulate) - change if your diagram uses different ADC pin

const unsigned long PUBLISH_INTERVAL_MS = 10000; // every 10 seconds

DHT dht(DHTPIN, DHTTYPE);
WiFiClient   espClient;
PubSubClient mqttClient(espClient);
unsigned long lastPublish = 0;

void connectWiFi() {
  Serial.print("Connecting to WiFi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 15000) {
    delay(500);
    Serial.print(".");
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi connected. IP: " + WiFi.localIP().toString());
  } else {
    Serial.println("\nWiFi connection failed — will retry.");
  }
}

void connectMQTT() {
  mqttClient.setServer(MQTT_BROKER, MQTT_PORT);
  String clientId = "iot-gdpr-" + String(DEVICE_ID) + "-" + String(random(0xFFFF), HEX);
  Serial.print("Connecting to MQTT broker " + String(MQTT_BROKER) + "...");
  unsigned long start = millis();
  while (!mqttClient.connected() && millis() - start < 10000) {
    if (mqttClient.connect(clientId.c_str())) {
      Serial.println(" connected.");
    } else {
      Serial.print(" failed (state=" + String(mqttClient.state()) + "), retrying...");
      delay(2000);
    }
  }
}

void publishTelemetry() {
  float temperature = dht.readTemperature();
  float humidity    = dht.readHumidity();

  int rawHR   = analogRead(HR_PIN);
  float heartRate = 50.0f + (rawHR / 4095.0f) * (180.0f - 50.0f);

  int rawSpO2 = analogRead(SPO2_PIN);
  float spo2  = 90.0f + (rawSpO2 / 4095.0f) * (100.0f - 90.0f);

  int rawResp = analogRead(RESP_PIN);
  int respRate = 10 + (int)((rawResp / 4095.0f) * (30 - 10));

  int rawBpSys = analogRead(BPSYS_PIN);
  int rawBpDia = analogRead(BPDIA_PIN);
  int bpSys = 90 + (int)((rawBpSys / 4095.0f) * (140 - 90));
  int bpDia = 60 + (int)((rawBpDia / 4095.0f) * (90 - 60));

  StaticJsonDocument<256> doc;
  doc["deviceToken"] = DEVICE_TOKEN;
  doc["deviceId"]    = DEVICE_ID;

  if (!isnan(temperature)) doc["temperature"] = round(temperature * 10.0) / 10.0;
  if (!isnan(humidity))    doc["humidity"]    = round(humidity * 10.0) / 10.0;
  doc["heartRate"] = (int)heartRate;
  doc["spo2"]      = (int)spo2;
  doc["respRate"]  = respRate;
  doc["bpSys"]     = bpSys;
  doc["bpDia"]     = bpDia;

  char payload[256];
  serializeJson(doc, payload);

  Serial.println("Publishing: " + String(payload));

  if (mqttClient.publish(MQTT_TOPIC, payload)) {
    Serial.println("✓ Published to " + String(MQTT_TOPIC));
  } else {
    Serial.println("✗ Publish failed");
  }
}

void setup() {
  Serial.begin(115200);
  delay(100);
  dht.begin();

  Serial.println("\n=== IoT GDPR Healthcare — Wokwi ESP32 (updated) ===");
  connectWiFi();
  connectMQTT();
}

void loop() {
  if (!mqttClient.connected()) {
    Serial.println("MQTT disconnected — reconnecting...");
    connectMQTT();
  }
  mqttClient.loop();

  if (WiFi.status() != WL_CONNECTED) {
    connectWiFi();
  }

  unsigned long now = millis();
  if (now - lastPublish >= PUBLISH_INTERVAL_MS) {
    lastPublish = now;
    publishTelemetry();
  }
}
